## 用途与范围

FP16 是 PTO 已分配的数值格式。本页帮助读者把载体布局、有限值分解和值分类联系起来；确切契约仍由 ASL 所有者中的 `FP16NumericFormatDescriptor`、`FP16FiniteDecomposition` 和 `ClassifyFP16` 定义。

<!-- PTO-READER-BLOCK: arch-format-fp16-concepts role=concepts-state -->
## 载体与字段

描述符使用 `16` 位载体、`16` 位逻辑 lane，并且每个载体包含 `1` 个 lane。lane 的符号位位于 `15`，具有 `5` 个指数位、`10` 个尾数位，指数偏置为 `15`。

载体中除符号、指数和尾数字段以外，没有额外的约束位。

<!-- PTO-READER-BLOCK: arch-format-fp16-rules role=rules-interactions -->
## 分解与分类

`FP16FiniteDecomposition` 先报告有限值分解是否可用；可用时返回符号、整数有效数和二进制指数。`ClassifyFP16` 则独立地把原始 lane 归入该格式所支持的零、次正规数、正规数、无穷大或 NaN 类别。

它具有带符号零、次正规数、无穷大、静默 NaN 和信号 NaN。全 1 指数不提供有限值分解；尾数为零时表示无穷大，尾数非零时进入 NaN 类别。

<!-- PTO-READER-BLOCK: arch-format-fp16-boundaries role=boundaries -->
## 边界与确切编码

`0x0000` 和 `0x8000` 分别是正零与负零；`0x7e00` 是规范静默 NaN。

有限值分解函数在返回的元组中报告可用性；值分类是另一个函数的独立结果。

pure func FP16FiniteDecomposition(value: bits(16))
    => (boolean, boolean, Word, integer {-1074..1023})
begin
    let exponent = value[14:10];
    let fraction = value[9:0];
    if exponent == Ones{5} then
        return (FALSE, FALSE, Zeros{PTO_XLEN}, 0);
    elsif exponent == Zeros{5} then
        if fraction == Zeros{10} then
            return (TRUE, value[15] == '1', Zeros{PTO_XLEN}, 0);
        else return (TRUE, value[15] == '1',
                     ZeroExtend{PTO_XLEN}(fraction), -24);
        end;
    else return (TRUE, value[15] == '1',
                 LSL(Zeros{PTO_XLEN} + 1, 10) +
                     ZeroExtend{PTO_XLEN}(fraction),
                 (UInt(exponent) - (15 + 10))
                     as integer {-1074..1023});
    end;
end;
pure func ClassifyFP16(value: bits(16)) => NumericValueClass
begin
    let exponent = value[14:10];
    let fraction = value[9:0];
    if exponent == Ones{5} then
        if fraction == Zeros{10} then
            if value[15] == '1' then return NumericValue_NegativeInfinity;
            else return NumericValue_PositiveInfinity;
            end;
        elsif fraction[9] == '1' then return NumericValue_QuietNaN;
        else return NumericValue_SignalingNaN;
        end;
    end;
    return NumericValueClassFromFiniteSign(value[15],
        exponent == Zeros{5} && fraction == Zeros{10},
        exponent == Zeros{5} && fraction != Zeros{10});
end;

pure func FP16CanonicalNaN() => Word
begin
    return Zeros{PTO_XLEN} + 0x7e00;
end;
